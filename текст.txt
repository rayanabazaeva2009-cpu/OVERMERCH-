        // Рендер карточек каталога
        function renderProducts() {
            const grid = document.getElementById('products-grid');
            const emptyState = document.getElementById('empty-state');
            const countEl = document.getElementById('products-count');

            // Фильтрация товаров
            const filtered = PRODUCTS.filter(product => {
                // Фильтр по категории / статусу
                if (state.category === 'musicians' && product.category !== 'musicians') return false;
                if (state.category === 'actors' && product.category !== 'actors') return false;
                if (state.category === 'in-stock' && product.status !== 'in-stock') return false;
                if (state.category === 'preorder' && product.status !== 'preorder') return false;

                // Фильтр по артисту
                if (state.artist !== 'Все артисты' && product.tag !== state.artist) return false;

                // Фильтр по поиску
                if (state.search.trim() !== '') {
                    const q = state.search.toLowerCase();
                    const matchArtist = product.artist.toLowerCase().includes(q);
                    const matchTitle = product.title.toLowerCase().includes(q);
                    const matchType = product.type.toLowerCase().includes(q);
                    if (!matchArtist && !matchTitle && !matchType) return false;
                }

                return true;
            });

            countEl.textContent = filtered.length;

            if (filtered.length === 0) {
                grid.innerHTML = '';
                emptyState.classList.remove('hidden');
                return;
            }

            emptyState.classList.add('hidden');

            grid.innerHTML = filtered.map(product => {
                const selectedSize = state.selectedSizes[product.id];
                const isInStock = product.status === 'in-stock';

                return `
                    <article class="bg-brand-card border border-brand-border hover:border-brand-steelLight/70 rounded-2xl overflow-hidden flex flex-col justify-between transition-all duration-300 group hover:-translate-y-1 shadow-lg">
                        
                        <!-- Верхняя зона с визуалом и бейджами -->
                        <div>
                            <div onclick="openProductModal(${product.id})" class="relative h-52 bg-gradient-to-br ${product.gradient} flex items-center justify-center p-6 cursor-pointer overflow-hidden border-b border-brand-border/60">
                                <!-- Фоновый водяной знак -->

                                    OVERMERCH
                                </div>

                                <span class="absolute top-3 left-3 px-2.5 py-1 rounded-md bg-brand-red text-white text-[10px] font-extrabold tracking-wider uppercase shadow">
                                    ОРИГИНАЛ
                                </span>

                                <!-- Бейдж статуса -->
                                <span class="absolute top-3 right-3 px-2.5 py-1 rounded-md text-[11px] font-semibold backdrop-blur-md border ${
                                    isInStock 
                                    ? 'bg-emerald-950/80 text-emerald-300 border-emerald-500/40' 
                                    : 'bg-brand-bg/85 text-brand-steelLight border-brand-steel/40'
                                }">
                                    ${product.statusLabel}
                                </span>

                                <!-- Векторный силуэт мерча -->
                                ${renderProductVisual(product)}

                                <!-- Подсказка быстрого просмотра -->
                                <span class="absolute bottom-2.5 right-3 text-[11px] text-brand-muted opacity-0 group-hover:opacity-100 transition-opacity bg-brand-bg/80 px-2 py-0.5 rounded">
                                    Подробнее →
                                </span>
                            </div>

                            <!-- Информация о товаре -->
                            <div class="p-5 pb-3">
                                <div class="flex items-center justify-between gap-2 mb-1">
                                    <span class="text-xs font-bold text-brand-steelLight uppercase tracking-wider truncate">
                                        ${product.artist}
                                    </span>
                                    <span class="text-[11px] text-brand-muted shrink-0">${product.type}</span>

        function resetFilters() {
            state.category = 'all';
            state.artist = 'Все артисты';
            state.search = '';
            document.getElementById('header-search').value = '';
            document.getElementById('mobile-search').value = '';

            });
        });

            document.getElementById('header-search').value = '';

            } else {
                overlay.classList.add('hidden');
                drawer.classList.add('translate-x-full');
            }
        }

        function addToCart(productId) {
            const product = PRODUCTS.find(p => p.id === productId);
            if (!product) return;

            const size = state.selectedSizes[productId];
            const existing = state.cart.find(item => item.productId === productId && item.size === size);

            if (existing) {
                existing.qty += 1;
            } else {
                state.cart.push({ productId, size, qty: 1 });
            }

            updateCartUI();
            showToast(`Добавлено: ${product.artist} (${size})`);
        }

        function changeCartQty(index, delta) {
            state.cart[index].qty += delta;
            if (state.cart[index].qty <= 0) {
                state.cart.splice(index, 1);
            }
            updateCartUI();
        }

        function removeCartItem(index) {
            state.cart.splice(index, 1);
            updateCartUI();
        }

        function updateCartUI() {
            const totalItems = state.cart.reduce((sum, item) => sum + item.qty, 0);
            document.getElementById('cart-badge').textContent = totalItems;
            document.getElementById('drawer-cart-count').textContent = totalItems;

            const container = document.getElementById('cart-items-container');
            const footer = document.getElementById('cart-footer');

            if (state.cart.length === 0) {
                footer.classList.add('hidden');
                container.innerHTML = `
                    <div class="h-full flex flex-col items-center justify-center text-center py-12">
                        <div class="w-16 h-16 rounded-full bg-brand-surface border border-brand-border flex items-center justify-center text-brand-muted mb-4">
                            <svg class="w-8 h-8" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                                <path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M16 11V7a4 4 0 00-8 0v4M5 9h14l1 12H4L5 9z"/>
                            </svg>
                        </div>
                        <div class="text-lg font-bold text-white mb-1">Ваша корзина пуста</div>
                        <p class="text-xs text-brand-muted max-w-xs mb-6">Добавьте оригинальный мерч из каталога, чтобы оформить доставку по России.</p>
                        <button onclick="toggleCart(false)" class="px-5 py-2.5 rounded-xl bg-brand-card border border-brand-border hover:border-brand-steelLight text-sm font-semibold text-white">
                            Перейти в каталог
                        </button>
                    </div>
                `;
                return;
            }

            footer.classList.remove('hidden');

            let subtotal = 0;
            container.innerHTML = state.cart.map((item, idx) => {
                const product = PRODUCTS.find(p => p.id === item.productId);
                const itemTotal = product.price * item.qty;
                subtotal += itemTotal;

                return `
                    <div class="bg-brand-card border border-brand-border rounded-2xl p-4 flex gap-3.5 items-center">
                        <div class="w-16 h-16 rounded-xl bg-gradient-to-br ${product.gradient} border border-brand-border/60 flex items-center justify-center shrink-0">
                            <span class="text-xs font-black text-white">${product.type.slice(0, 4).toUpperCase()}</span>
                        </div>
                        <div class="flex-1 min-w-0">
                            <div class="text-[11px] font-bold text-brand-steelLight uppercase truncate">${product.artist}</div>
                            <div class="text-sm font-bold text-white truncate">${product.title}</div>
                            <div class="text-xs text-brand-muted mt-0.5">Размер: <span class="text-white font-semibold">${item.size}</span></div>
                            <div class="flex items-center justify-between mt-2">
                                <div class="text-sm font-extrabold text-white">${formatPrice(itemTotal)}</div>
                                <div class="flex items-center gap-2 bg-brand-bg border border-brand-border rounded-lg px-2 py-0.5">
                                    <button onclick="changeCartQty(${idx}, -1)" class="text-brand-muted hover:text-white font-bold px-1">−</button>
                                    <span class="text-xs font-bold text-white min-w-[14px] text-center">${item.qty}</span>
                                    <button onclick="changeCartQty(${idx}, 1)" class="text-brand-muted hover:text-white font-bold px-1">+</button>
                                </div>
                            </div>
                        </div>
                        <button onclick="removeCartItem(${idx})" class="text-brand-muted hover:text-brand-red p-1 self-start" title="Удалить">
                            ✕
                        </button>
                    </div>
                `;
            }).join('');

            const shipping = parseInt(document.getElementById('delivery-method').value, 10);
            document.getElementById('cart-subtotal').textContent = formatPrice(subtotal);
            document.getElementById('cart-shipping').textContent = formatPrice(shipping);
            document.getElementById('cart-total').textContent = formatPrice(subtotal + shipping);
        }

        // Оформление заказа из корзины
        function submitOrder() {
            const contactInput = document.getElementById('checkout-contact');
            const contact = contactInput.value.trim();

            if (!contact) {
                contactInput.classList.add('border-brand-red');
                showToast('Укажите Telegram или телефон для оформления заказа', true);
                contactInput.focus();
                return;
            }

            contactInput.classList.remove('border-brand-red');
            const orderNumber = Math.floor(10000 + Math.random() * 90000);
            state.cart = [];
            contactInput.value = '';
            updateCartUI();
            toggleCart(false);

            showToast(`Заказ #OM-${orderNumber} оформлен! Менеджер свяжется с вами (${contact})`);
        }

        // Обработка формы Консьерж-сервиса
        document.getElementById('concierge-form').addEventListener('submit', function(e) {
            e.preventDefault();
            const item = document.getElementById('concierge-item').value.trim();
            const contact = document.getElementById('concierge-contact').value.trim();

            const successBox = document.getElementById('concierge-success');
            const successText = document.getElementById('concierge-success-text');

            successText.textContent = `Мы получили запрос на «${item}». Наш байер проверит наличие в официальном зарубежном сторе и напишет вам в ${contact} в течение 15 минут.`;
            successBox.classList.remove('hidden');

            showToast('Заявка на индивидуальный выкуп отправлена!');
            this.reset();
        });

        // Всплывающее уведомление (Toast)
        function showToast(message, isError = false) {
            const container = document.getElementById('toast-container');
            const toast = document.createElement('div');
            toast.className = `pointer-events-auto p-4 rounded-xl border shadow-2xl backdrop-blur-md text-sm font-medium flex items-center gap-3 transition-all duration-300 ${
                isError 
                ? 'bg-red-950/90 border-brand-red text-white' 
                : 'bg-brand-card/95 border-brand-steelLight/60 text-white'
            }`;
            toast.innerHTML = `
                <span class="w-2.5 h-2.5 rounded-full shrink-0 ${isError ? 'bg-brand-red' : 'bg-emerald-400'}"></span>
                <span>${message}</span>
            `;
            container.appendChild(toast);

            setTimeout(() => {
                toast.style.opacity = '0';
                toast.style.transform = 'translateY(10px)';
                setTimeout(() => toast.remove(), 300);
            }, 3500);
        }
